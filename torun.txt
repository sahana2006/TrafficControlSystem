# Emergency Vehicle Traffic Control — run commands

# 1. Install
pip install -r requirements.txt

# 2. Set SUMO (Windows PowerShell)
$env:SUMO_HOME = "C:\Program Files (x86)\Eclipse\Sumo"
$env:PATH = "$env:SUMO_HOME\bin;$env:PATH"

GUI + default logs + graphs
python main.py --gui

Heavier traffic (700 vehicles)
python main.py --gui --traffic high

Run for 1200 seconds
python main.py --gui --end 1200

General log every 3 s
python main.py --gui --log-interval 3

EV position/speed log every 1 s
python main.py --gui --ev-log-interval 1

GUI + logs only, no graphs
python main.py --gui --no-plots

GUI + frequent EV logs + medium traffic
python main.py --gui --traffic medium --end 900 --log-interval 5 --ev-log-interval 1
